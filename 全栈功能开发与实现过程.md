CREATE TABLE polysmart_strategy_config (

## 三、 核心定价与套利引擎 (Master Node)

Master 节点负责订阅异构市场行情、运行 OU 均值回归并根据用户设定的 αfloor​ 实时产出多账号分批挂单指导价。

Python

```
# master_engine.py
import asyncio
import numpy as np
from dataclasses import dataclass

@dataclass
class MarketData:
    poly_mid: float
    kalshi_mid: float
    poly_vol: float
    kalshi_vol: float

class PolysmartMasterEngine:
    def __init__(self, market_id, db_pool):
        self.market_id = market_id
        self.db = db_pool
        self.ou_mu = 0.0020        # OU 长期均值价差项
        self.ou_theta = 2.15       # 均值回归速度系数
        
    async def fetch_strategy_params(self):
        """实时从数据库拉取用户调整的保底收益率"""
        async with self.db.acquire() as conn:
            row = await conn.fetchrow(
                "SELECT alpha_floor, frictions_base FROM polysmart_strategy_config WHERE market_id = $1", 
                self.market_id
            )
            return float(row['alpha_floor']), float(row['frictions_base'])

    def calculate_adaptive_alpha(self, current_spread, alpha_floor, sigma):
        """
        核心数学模型：联立保底收益率与价格偏离度
        当市场严重撕裂时，推高预期溢价；平时死守用户保底底线
        """
        ou_premium = self.ou_theta * abs(current_spread - self.ou_mu)
        vol_premium = 1.96 * sigma  # 95% 置信度波动溢价
        
        # 动态期望收益目标
        alpha_target = max(alpha_floor, ou_premium + vol_premium)
        return alpha_target

    async def generate_quoting_signals(self, market_data: MarketData, total_inventory: float):
        """生成具备全局库存惩罚的双端挂单指导价"""
        alpha_floor, frictions = await self.fetch_strategy_params()
        
        # 1. 计算当前两端合成中间价与瞬时价差
        synthetic_mid = (market_data.poly_mid + market_data.kalshi_mid) / 2
        current_spread = market_data.poly_mid - market_data.kalshi_mid
        
        # 2. 结合瞬时整体波动率计算自适应阿尔法目标
        combined_sigma = np.sqrt(market_data.poly_vol**2 + market_data.kalshi_vol**2)
        alpha_target = self.calculate_adaptive_alpha(current_spread, alpha_floor, combined_sigma)
        
        # 3. 引入全局库存惩罚机制 (对 Reservation Price 进行倾斜)
        gamma = 0.12  # 库存风险厌恶系数
        reservation_price = synthetic_mid - (total_inventory * gamma * (combined_sigma**2))
        
        # 4. 推导两端无风险锁仓路径的最终挂单限价 (Maker 单)
        total_buffer = alpha_target + frictions
        poly_bid_yes = reservation_price - (total_buffer / 2)
        kalshi_bid_no = (1.0 - reservation_price) - (total_buffer / 2)
        
        return {
            "status": "EXECUTE" if poly_bid_yes > 0 and kalshi_bid_no > 0 else "WAIT",
            "poly_target_bid": round(poly_bid_yes, 3),
            "kalshi_target_bid": round(kalshi_bid_no, 3),
            "alpha_deployed": alpha_target
        }
```

## 四、 多账号分布式执行与反作弊隔离层 (Slave Worker)

Slave 节点集群接收到 Master 的价格信号后，通过分单路由算法（Order Slicer）进行非均匀打散，并绑定独立住宅代理 IP 发射订单。

Python

```
# slave_worker.py
import asyncio
import random
import aiohttp
from time import time

class PolysmartSlaveWorker:
    def __init__(self, account_id, platform, proxy_url, credentials):
        self.account_id = account_id
        self.platform = platform
        self.proxy_url = proxy_url
        self.creds = credentials
        
        # 限频自适应状态
        self.last_action_time = 0.0
        self.min_interval = 0.25  # 基础 250ms 防御栅栏
        self.current_order_id = None

    def _apply_anti_sybil_jitter(self, base_price, base_volume):
        """反作弊混淆机制：对价格和数量施加非均匀微幅扰动，破坏集群行为共性"""
        jittered_price = base_price - random.uniform(0.001, 0.003)
        jittered_volume = base_volume * random.uniform(0.85, 1.15)
        return round(jittered_price, 3), round(jittered_volume, 1)

    async def execute_layered_quote(self, target_price, total_allocated_vol, side="BUY"):
        """带隔离代理的非对称挂单逻辑"""
        now = time()
        if now - self.last_action_time < self.min_interval:
            return  # 触发硬性时间缓冲保护，直接拒绝高频调用
            
        final_price, final_vol = self._apply_anti_sybil_jitter(target_price, total_allocated_vol)
        
        # 封装独立住宅代理头
        connector = aiohttp.ProxyConnector(proxy_url=self.proxy_url) if self.proxy_url else None
        async with aiohttp.ClientSession(connector=connector) as session:
            # 根据底层平台调用异构网关
            if self.platform == "kalshi":
                await self._submit_kalshi_order(session, final_price, final_vol, side)
            elif self.platform == "polymarket":
                await self._submit_poly_order(session, final_price, final_vol, side)
                
        self.last_action_time = time()

    async def _submit_kalshi_order(self, session, price, volume, side):
        """调用 Kalshi 的 REST 变更加签接口 (使用其自带的 Modify 以省配额)"""
        url = f"https://api.kalshi.com/v2/portfolio/orders"
        headers = {"Authorization": f"Bearer {self.creds['token']}", "Content-Type": "application/json"}
        payload = {
            "action": side, "count": int(volume), "type": "limit",
            "yes_price": int(price * 100), "ticker": "FED-26JUN-B"
        }
        # 生产环境下此处加入带有 API 签名鉴权的加密机制
        async with session.post(url, json=payload, headers=headers) as resp:
            if resp.status == 201:
                data = await resp.json()
                self.current_order_id = data.get("order_id")
            elif resp.status == 429:
                self.min_interval *= 1.5  # 遭遇 429 指数级退避拉长静默缓冲

    async def _submit_poly_order(self, session, price, volume, side):
        """调用 Polymarket 链下撮合 CLOB 接口"""
        # Web3 挂单免除 250ms Taker 惩罚延迟，直接异步投递
        pass 
```

### 异步跨网关对冲状态机实现

当矩阵中任意一个 Slave 账号被外部订单撞单成交时，系统必须在 **800 毫秒** 强制时限内指挥对侧进行对冲平锁：

Python

```
# hedging_coordinator.py
class DistributedHedgeCoordinator:
    def __init__(self, master_engine, slaves_matrix):
        self.master = master_engine
        self.slaves = slaves_matrix # 包含所有运行中账号实体的字典
        self.hedge_timeout = 0.8     # 800ms 强制锁仓限期

    async def handle_fill_event(self, trigger_account_id, platform, fill_qty, event_market):
        """任何一端 Slave 产生 Fill 状态改变的异步回调处理器"""
        print(f"【⚠️ 矩阵成交通知】 账户: {trigger_account_id} 侧: {platform} 数量: {fill_qty}")
        
        # 1. 挂起全局主挂单器，暂停更新限价盘，防止资金竞争
        self.master.is_hedging = True
        
        # 2. 定位对冲目标阵营
        target_platform = "kalshi" if platform == "polymarket" else "polymarket"
        
        # 3. 筛选对侧可用资产最为健康且未触碰持仓上限的 Slave 账户
        candidate_slaves = [
            s for s in self.slaves.values() 
            if s.platform == target_platform and s.status == "healthy"
        ]
        
        if not candidate_slaves:
            raise SystemCriticalException("全矩阵对侧对冲通道资金断流！触发全局报警。")
            
        selected_slave = candidate_slaves[0]
        
        # 4. 双轨异步对冲推进：先尝试 Maker 撮合，若超时立马转 Taker 强吃
        start_hedge_t = time()
        success = False
        
        # 投递对冲 Maker 单
        await selected_slave.execute_layered_quote(target_price=0.48, total_allocated_vol=fill_qty, side="BUY")
        
        while time() - start_hedge_t < self.hedge_timeout:
            if await selected_slave.check_fill_complete():
                success = True
                print("【🎉 完美对冲】对侧分布式账号 Maker 顺利成交，无风险套利利润已锁定！")
                break
            await asyncio.sleep(0.05)
            
        if not success:
            # 状态机强行越界：撤销原 Maker 单，化身 Taker 砸盘强行吃掉对侧订单簿深度
            print("【🚨 对冲时限耗尽】对侧未能在 800ms 内成交，启动 Taker 强制市价锁仓...")
            await selected_slave.cancel_active_order()
            await selected_slave.execute_emergency_taker(fill_qty)
            
        # 5. 释放对冲锁，更新全局库存，主流程复位
        self.master.is_hedging = False
```

## 六、 生产环境全栈调试与部署指南

1. **分布式环境对时：** 由于多账号高频撮合强依赖时间戳，Master 节点与 Slave 节点部署服务器必须开通 **Chrony (NTP) 高精度纳秒级时间同步服务**，确保本地时钟与纽交所/美东 AWS 骨干网时钟偏离度 <1ms。
    
2. **Web2 网关反爬升级防护：** Kalshi 和 PredictIt 会不定期升级 Cloudflare WAF（防机器人校验）。生产环境部署时，需将 Slave 驱动层封装进带有 **TLS 指纹伪装（如 JA3/JA4 指纹修改器）** 的定制化 Python 请求库中，使 Agent 网络的每一次 API 请求从底层 TCP 握手层面完美模拟原生的 Chrome 浏览器行为。
